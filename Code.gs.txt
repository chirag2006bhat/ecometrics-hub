function getDashboardData() {
  // 🚀 CACHE BUSTER: Forcing the AI to forget old data and read fresh!
  const cache = CacheService.getScriptCache();
  const cachedData = cache.get("ecoMetricsDataCacheV4");
  if (cachedData) return JSON.parse(cachedData);

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  const detailsSheet = ss.getSheetByName("Location Details");
  if (!detailsSheet) throw new Error("Could not find 'Location Details' sheet.");
  
  const detailsData = detailsSheet.getDataRange().getValues();
  const locationsMap = {}; 
  
  const headers = detailsData[0];
  const latIdx = headers.indexOf("Lat");
  const lngIdx = headers.indexOf("Lng");
  const nameIdx = headers.indexOf("Name of location");
  const codeIdx = headers.indexOf("Location Code");
  
  // Grabs the Source Type and Testing Type columns directly from your spreadsheet
  const sourceIdx = headers.indexOf("Source Type"); 
  const testingTypeIdx = headers.indexOf("Testing Type");

  for (let i = 1; i < detailsData.length; i++) {
    let code = detailsData[i][codeIdx];
    let locName = detailsData[i][nameIdx];
    if (!code) continue; 
    
    let sheetName = String(code).replace("Location -", "Location-").trim();
    // Extract the source type (defaults to "Other" if staff left it blank)
    let srcType = (sourceIdx > -1 && detailsData[i][sourceIdx]) ?
        String(detailsData[i][sourceIdx]).trim() : "Other / Unspecified";
    let tType = (testingTypeIdx > -1 && detailsData[i][testingTypeIdx]) ? 
        String(detailsData[i][testingTypeIdx]).trim() : "Water Testing";

    let lat = (latIdx > -1 && detailsData[i][latIdx]) ?
        detailsData[i][latIdx] : 13.3300 + (Math.random() * 0.05);
    let lng = (lngIdx > -1 && detailsData[i][lngIdx]) ?
        detailsData[i][lngIdx] : 77.1000 + (Math.random() * 0.05);

    locationsMap[sheetName] = { name: locName, lat: parseFloat(lat), lng: parseFloat(lng), sourceType: srcType, testingType: tType, records: {} };
  }

  let allParameters = new Set();
  let allMonths = new Set();
  const sheets = ss.getSheets();
  function parseDate(val) {
      if (val instanceof Date) return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd");
      return String(val).trim();
  }
  
  sheets.forEach(sheet => {
    let sheetName = sheet.getName().trim();
    if (locationsMap[sheetName]) {
      let data = sheet.getDataRange().getValues();
      if (data.length < 4) return; // Skip if sheet doesn't have enough rows

      let typeIdentifier = String(data[1][0]).trim().toLowerCase(); 

      // 1. STANDARD LAYOUT (Locations 1-36)
      if (typeIdentifier === "month" || typeIdentifier === "date") {
         let paramHeaders = data[2]; // Parameters are in Row 3
 
         for (let c = 1; c < paramHeaders.length; c++) {
           let p = String(paramHeaders[c]).trim();
           if(p) allParameters.add(p);
         }

         for (let r = 3; r < data.length; r++) { // Data starts in Row 4
           let rawDate = data[r][0]; 
         
           if (!rawDate || String(rawDate).trim() === "") continue;
           
           let monthStr = parseDate(rawDate);
           if (!monthStr || monthStr === "-" || monthStr.toLowerCase() === "average") continue;
           allMonths.add(monthStr);
           if (!locationsMap[sheetName].records[monthStr]) locationsMap[sheetName].records[monthStr] = {};

           for (let c = 1; c < paramHeaders.length; c++) {
             let pName = String(paramHeaders[c]).trim();
             if (pName) locationsMap[sheetName].records[monthStr][pName] = data[r][c];
           }
         }
      } 
      // 2. TRANSPOSED STP LAYOUT (Locations 37-38)
      else if (typeIdentifier === "parameter") {
         let dateHeaders = data[2];
         // Dates are in Row 3
         
         for (let c = 1; c < dateHeaders.length; c++) {
           let rawDate = dateHeaders[c];
           if (!rawDate) continue;
           let m = parseDate(rawDate);
           if (m && m !== "-" && m.toLowerCase() !== "average") allMonths.add(m);
         }

         for (let r = 3; r < data.length; r++) { // Data starts in Row 4
           let pName = String(data[r][0]).trim();
           if (!pName) continue;

           allParameters.add(pName);

           for (let c = 1; c < dateHeaders.length; c++) {
             let rawDate = dateHeaders[c];
             if (!rawDate) continue;
             let monthStr = parseDate(rawDate);
             if (!monthStr || monthStr === "-" || monthStr.toLowerCase() === "average") continue;
             if (!locationsMap[sheetName].records[monthStr]) locationsMap[sheetName].records[monthStr] = {};
             locationsMap[sheetName].records[monthStr][pName] = data[r][c];
           }
         }
      }
    }
  });
  
  // --- DYNAMIC THRESHOLD READER ---
  let dynamicThresholds = { "Drinking Water": {}, "STP": {}, "Surface Water": {} };
  const thresholdSheet = ss.getSheetByName("Thresholds");
  
  if (thresholdSheet) {
    const tData = thresholdSheet.getDataRange().getValues();
    const tHeaders = tData[0].map(h => String(h).trim().toLowerCase());
    const typeIdx = tHeaders.findIndex(h => h.includes("testing type"));
    
    // Start at i=1 to skip the header row
    for (let i = 1; i < tData.length; i++) {
      let param = String(tData[i][0]).trim();
      let category = String(tData[i][1]).trim();
      
      let minVal = tData[i][2] === "" ? undefined : parseFloat(tData[i][2]);
      let maxVal = tData[i][3] === "" ? undefined : parseFloat(tData[i][3]);
      // NEW: Read the 5th column to check if it's a strict parameter
      let strictCell = String(tData[i][4]).trim().toLowerCase();
      let isStrict = (strictCell === "yes" || strictCell === "true" || strictCell === "y");
      let tType = (typeIdx > -1 && tData[i][typeIdx]) ? String(tData[i][typeIdx]).trim() : "Water Testing";
      
      if (param && category) {
        if (!dynamicThresholds[category]) dynamicThresholds[category] = {};
        // Save the strict status and testing type inside the parameter's data
        dynamicThresholds[category][param] = { strict: isStrict, testingType: tType };
        if (minVal !== undefined && !isNaN(minVal)) {
          dynamicThresholds[category][param].min = minVal;
        }
        if (maxVal !== undefined && !isNaN(maxVal)) {
          dynamicThresholds[category][param].max = maxVal;
        }
      }
    }
  }

  const finalData = {
    locations: Object.values(locationsMap),
    months: Array.from(allMonths).sort(),
    parameters: Array.from(allParameters).filter(p => p !== "Month" && p !== "Parameter"),
    thresholds: dynamicThresholds 
  };

  // 🚀 CACHE BUSTER: Save fresh data to V4
  try { cache.put("ecoMetricsDataCacheV4", JSON.stringify(finalData), 60); } catch(e) {}

  return finalData;

}
/**
 * Communicates with the Google Gemini API using the official request structure.
 * Stored inside Script Properties under the key: GEMINI_API_KEY
 * * @param {string} promptText The prompt sent to the model.
 * @return {string} The text response from Gemini.
 */
function callGeminiAPI(promptText) {
  const apiKey = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
  if (!apiKey) {
    throw new Error("API Key configuration error: 'GEMINI_API_KEY' is missing from Script Properties.");
  }

  const model = "gemini-3.1-flash-lite";
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;
  const payload = {
    contents: [
      {
        parts: [
          { text: promptText }
        ]
      }
    ]
  };
  const options = {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };
  try {
    const response = UrlFetchApp.fetch(url, options);
    const responseCode = response.getResponseCode();
    const responseText = response.getContentText();
    const json = JSON.parse(responseText);

    if (responseCode !== 200) {
      Logger.log("Gemini API Error Response: " + responseText);
      throw new Error(`Gemini API returned status code ${responseCode}: ${json.error ? json.error.message : responseText}`);
    }

    if (json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts && json.candidates[0].content.parts[0]) {
      return json.candidates[0].content.parts[0].text;
    } else {
      Logger.log("Unexpected Gemini Payload Format: " + responseText);
      throw new Error("Failed to extract valid text from Gemini API response structure.");
    }
  } catch (error) {
    Logger.log("Execution catch triggered in callGeminiAPI: " + error.toString());
    throw new Error("Backend connection failure: " + error.message);
  }
}

/**
 * Senior Developer Hybrid AI Router Engine - Master Final Corrected Edition
 * Processes chatbot queries against live database rows using exact dashboard threshold rules.
 * Ensures summaries, parameter checks, and general queries default to historical averages unless a date is specified.
 * * @param {string} userQuery Raw query string from front-end.
 * @return {string} Pure factual tabular metadata or controlled Gemini conceptual explanations.
 */
function handleChatbotQuery(userQuery, chatMemory = "", activeTestingType = "All") {
  if (!userQuery) return "Please enter a valid question.";
  
  const queryLower = userQuery.toLowerCase().trim();
  const queryNormalized = queryLower.replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ");
  
  // ==========================================
  // POINT 4 & 5: SECURITY & SANDBOX FILTERING
  // ==========================================
  const blockWords = ["apps script", "source code", "backend", "database", "developer prompt", "system prompt", "ignore previous instructions", "hidden prompt", "api key", "script properties", "code.gs", "index.html"];
  for (let word of blockWords) {
    if (queryLower.includes(word)) {
      return "Sorry, I cannot provide that information.";
    }
  }

  // ==========================================
  // INITIALIZE DATABASE READ
  // ==========================================
  let data;
  try {
    data = getDashboardData();
  } catch(e) {
    return "Error accessing spreadsheet records: " + e.message;
  }

  // Helper mapping function matching your front-end getZone logic perfectly
  // 1. PERFECT SYNCHRONIZATION: 1:1 Copy of Frontend Zone Logic
  function getMasterLimits(parameter, locationName) {
    const locData = data.locations.find(l => l.name === locationName);
    const sourceType = locData && locData.sourceType ? String(locData.sourceType).trim() : "Drinking Water";
    const tType = locData && locData.testingType ? String(locData.testingType).trim() : "Water Testing";
    
    let catKey = Object.keys(data.thresholds).find(k => k.toLowerCase() === sourceType.toLowerCase());
    if (!catKey) catKey = Object.keys(data.thresholds).find(k => k.toLowerCase() === tType.toLowerCase());
    if (!catKey) {
        const sTypeUp = sourceType.toUpperCase();
        if (sTypeUp.includes("STP") || sTypeUp.includes("SEWAGE")) catKey = Object.keys(data.thresholds).find(k => k.toLowerCase() === "stp");
        else if (sTypeUp.includes("LAKE") || sTypeUp.includes("POND") || sTypeUp.includes("RIVER") || sTypeUp.includes("SURFACE")) catKey = Object.keys(data.thresholds).find(k => k.toLowerCase() === "surface water");
        else catKey = Object.keys(data.thresholds).find(k => k.toLowerCase() === "drinking water");
    }

    let limits = null;
    if (catKey && data.thresholds[catKey]) {
        const paramKey = Object.keys(data.thresholds[catKey]).find(k => k.toLowerCase() === parameter.toLowerCase());
        if (paramKey) limits = data.thresholds[catKey][paramKey];
    }
    return limits;
  }

  function isZeroTolerance(parameter, locationName) {
    if (parameter.toLowerCase().includes("coli") || parameter.toLowerCase().includes("bacteria")) return true;
    let limits = getMasterLimits(parameter, locationName);
    if (limits && limits.strict === true && limits.min === undefined && limits.max === undefined) return true;
    return false;
  }

  
  function calculateZoneColor(parameter, value, locationName) {
    if (value === undefined || value === null || value === "") return "Grey";
    let cleanStr = String(value).trim().toLowerCase();
    if (cleanStr === "-" || cleanStr === "grey") return "Grey";
    if (cleanStr === "bdl" || cleanStr === "nd") cleanStr = "0";

    if (cleanStr.includes("absent") || cleanStr.includes("agreeable") || cleanStr.includes("not detectable") || cleanStr === "nil" || cleanStr.includes("free from")) return "Green";
    if (cleanStr.includes("present") || cleanStr.includes("disagreeable") || cleanStr.includes("objectionable") || cleanStr.includes("positive") || cleanStr.includes("detectable")) return "Red";

    const numValue = parseFloat(cleanStr.replace(/[^\d.-]/g, ''));
    if (isNaN(numValue)) return "Grey";

    let limits = getMasterLimits(parameter, locationName);
    if (!limits && !isZeroTolerance(parameter, locationName)) return "Green"; 

    let isRed = false, isYellow = false;
    let buffer = (limits && limits.strict === true) ? 0.0 : 0.10; 

    if (isZeroTolerance(parameter, locationName)) {
        if (numValue > 0) return "Red";
    } else if (limits) {
        if (limits.min !== undefined && numValue < limits.min) isRed = true;
        else if (limits.min !== undefined && buffer > 0 && numValue <= (limits.min + Math.abs(limits.min * buffer))) isYellow = true;
        
        if (limits.max !== undefined && numValue > limits.max) isRed = true;
        else if (limits.max !== undefined && buffer > 0 && numValue >= (limits.max - Math.abs(limits.max * buffer))) isYellow = true;
    }

    if (isRed) return "Red";
    if (isYellow) return "Yellow";
    return "Green";
  }

  // ==========================================
  // SMART LOCATION FUZZY REGISTRATION
  // ==========================================
  let matchedLocation = null;
  let bestScore = 0;

  for (let loc of data.locations) {
    const nameNorm = loc.name.toLowerCase().replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ");
    if (queryNormalized.includes(nameNorm) || queryLower.includes(loc.name.toLowerCase())) {
      matchedLocation = loc;
      break;
    }
  }

  if (!matchedLocation) {
    for (let loc of data.locations) {
      const nameLow = loc.name.toLowerCase();
      const uniqueTokens = nameLow.replace("pump house", "").replace("water purifier", "").replace("tank", "").replace("layout", "").split(/[\s\-]+/);
      let matchCount = 0;
      for (let token of uniqueTokens) {
        let cleanToken = token.trim();
        if (cleanToken.length >= 3) {
          if (queryNormalized.split(" ").includes(cleanToken)) {
            matchCount++;
          }
        }
      }
      if (matchCount > bestScore && matchCount >= 1) {
        bestScore = matchCount;
        matchedLocation = loc;
      }
    }
  }
  // CRITICAL SECURITY ISOLATION BOUNDARY (Smart Filter)
  const isGlobalAnalysisQuery = queryLower.includes("highest") || queryLower.includes("maximum") || queryLower.includes("max") || queryLower.includes("red zone") || queryLower.includes("yellow zone") || queryLower.includes("green zone") || queryLower.includes("all locations");
  
  if (!matchedLocation && !isGlobalAnalysisQuery) {
    // Blocks unknown places (like "USA") but ALLOWS pronouns ("this", "it") and dates ("March", "2024") to pass to Memory
    const blockRegex = /\b(water|quality|status|report|details|summary|info|ph|tds|value|values|metrics|hardness|level|levels|coliform|bacteria|iron|fluoride|nitrate|arsenic|turbidity|chlorine|parameter|data)\b\s+(of|in|for|at)\s+(?!this\b|it\b|here\b|there\b|the\b|that\b|those\b|these\b|all\b|january\b|february\b|march\b|april\b|may\b|june\b|july\b|august\b|september\b|october\b|november\b|december\b|jan\b|feb\b|mar\b|apr\b|aug\b|sept\b|oct\b|nov\b|dec\b|20\d{2}\b)[a-z0-9]+/i;
    
    if (queryLower.match(blockRegex)) {
       return "I can only answer questions related to the EcoMetrics Hub and locations present in the spreadsheet.";
    }
  }

  // --- NEW: SMART CONTEXTUAL MEMORY ---
  if (!matchedLocation && chatMemory && !isGlobalAnalysisQuery) {
    // ONLY trigger memory if they use a follow-up pronoun OR it's a short question. 
    // DO NOT trigger memory if they explicitly type a new, unknown location name after "in", "at", or "for".
    const isFollowUp = /\b(it|its|this|that|here|there|same)\b/i.test(queryLower) || queryLower.split(/\s+/).length <= 5;
    const hasUnknownLocationTrap = /\b(in|at|for|of)\s+(?!the\b|all\b|20\d{2}\b)[a-z]+/i.test(queryLower);

    if (isFollowUp && !hasUnknownLocationTrap) {
      let lastMentionedIndex = -1;
      let bestMemoryLoc = null;
      for (let loc of data.locations) {
        let idx = chatMemory.toLowerCase().lastIndexOf(loc.name.toLowerCase());
        if (idx > lastMentionedIndex) {
          lastMentionedIndex = idx;
          bestMemoryLoc = loc; 
        }
      }
      if (bestMemoryLoc) {
        matchedLocation = bestMemoryLoc;
      }
    }
  }
 // ==========================================
  // SMART LOCATION-AWARE PARAMETER SCANNER
  // ==========================================
  let foundParam = null;
  let isAllParams = queryLower.includes("all parameter") || queryLower.includes("all metrics") || queryLower.includes("all the parameter");
  if (!isAllParams) {
    let searchParams = data.parameters;
    // If we know the location, ONLY search parameters that actually exist for it
    if (matchedLocation) {
      let locParams = new Set();
      Object.values(matchedLocation.records).forEach(rec => Object.keys(rec).forEach(k => locParams.add(k)));
      searchParams = Array.from(locParams);
    }

    if (searchParams && searchParams.length > 0) {
      for (let actualP of searchParams) {
        let cleanParamName = actualP.toLowerCase().trim();
        let cleanParamBase = cleanParamName.replace(/,?\s*mg\/l/g, "").replace(/,?\s*per\s*\d+ml/g, "").replace(/,?\s*mpn\/\d+\s*ml/g, "").trim();

        // Special precise matching for short names like "pH" and handling base names
        if (queryLower.includes(cleanParamName) || 
           (cleanParamBase.length >= 3 && queryLower.includes(cleanParamBase)) || 
           (queryLower.match(/\bph\b/) && cleanParamName.includes("ph"))) {
          foundParam = actualP;
          break; 
        }
      }
    }
  }
 // ==========================================
  // DATE PARSING ENGINE
  // ==========================================
  let targetMonth = null;
  const monthNames = ["january", "february", "march", "april", "may", "june", "july", "august", "september", "october", "november", "december"];
  const shortNames = ["jan", "feb", "mar", "apr", "may", "jun", "jul", "aug", "sep", "oct", "nov", "dec"];

  for (let m of data.months) {
    const shortMonth = m.substring(0, 7);
    if (queryLower.includes(m) || queryLower.includes(shortMonth)) {
      targetMonth = m;
      break;
    }

    const parts = m.split("-");
    if (parts.length >= 2) {
      const yearStr = parts[0];
      const monthIdx = parseInt(parts[1], 10) - 1;
      if (monthIdx >= 0 && monthIdx < 12) {
        const verbalFull = monthNames[monthIdx] + " " + yearStr;
        const verbalShort = shortNames[monthIdx] + " " + yearStr;
        if (queryLower.includes(verbalFull) || queryLower.includes(verbalShort)) {
          targetMonth = m;
          break;
        }
      }
    }
  }

  // 🛑 MISSING DATA TRAP: Catch years/dates that don't exist in the spreadsheet
  const yearMatch = queryLower.match(/\b(20\d{2})\b/); 
  const monthMatch = queryLower.match(/\b(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|jun|jul|aug|sep|sept|oct|nov|dec)\b/);
  
  let askedForTime = yearMatch ? yearMatch[0] : (monthMatch ? monthMatch[0] : null);

  if (askedForTime && !targetMonth && !queryLower.includes("average") && !queryLower.includes("history")) {
      let timeExistsInRecords = data.months.some(m => {
          if (m.toLowerCase().includes(askedForTime)) return true;
          let parts = m.split("-");
          if (parts.length >= 2) {
               let mIdx = parseInt(parts[1], 10) - 1;
               if (mIdx >= 0 && mIdx < 12 && ((monthNames[mIdx] + " " + parts[0]).includes(askedForTime) || (shortNames[mIdx] + " " + parts[0]).includes(askedForTime))) return true;
          }
          return false;
      });
      
      if (!timeExistsInRecords) {
          return `I'm sorry, but there is no valid data recorded for "${askedForTime}" in this dashboard. I can only provide data for the specific time periods listed in the dropdown menu.`;
      }
  }

  // ==========================================
  // POINT 7: DYNAMIC ZONES MODULE (Perfectly Synced with Map Math)
  // ==========================================
  const isRedQuery = queryLower.includes("red zone") || queryLower.includes("redzone") || queryLower.includes("violation") || queryLower.includes("failing") || queryLower.includes("non-compliant");
  const isYellowQuery = queryLower.includes("yellow zone") || queryLower.includes("warning");
  const isGreenQuery = queryLower.includes("green zone") || queryLower.includes("compliant") || queryLower.includes("safe location");

  if (isRedQuery || isYellowQuery || isGreenQuery) {
    let redLocations = new Set();
    let yellowLocations = new Set();
    let greenLocations = new Set();

    let targetLocations = data.locations;
    if (activeTestingType && activeTestingType !== "All") {
        targetLocations = data.locations.filter(l => (l.testingType || "Water Testing") === activeTestingType);
    }

    targetLocations.forEach(loc => {
      let finalLocColor = "Grey";
      let hasDataPoints = false;

      if (targetMonth) {
        if (loc.records[targetMonth]) {
          data.parameters.forEach(pKey => {
            let val = loc.records[targetMonth][pKey];
            if (val !== undefined && val !== "" && String(val).trim() !== "-") {
                let color = calculateZoneColor(pKey, val, loc.name);
                if (color !== "Grey") hasDataPoints = true;
                
                if (color === "Red") finalLocColor = "Red";
                else if (color === "Yellow" && finalLocColor !== "Red") finalLocColor = "Yellow";
                else if (color === "Green" && finalLocColor === "Grey") finalLocColor = "Green";
            }
          });
        }
      } else {
        data.parameters.forEach(pKey => {
          let sum = 0, count = 0, textValues = [];
          let violationCount = 0;
          
          Object.keys(loc.records).forEach(mKey => {
            let rawVal = loc.records[mKey][pKey];
            if (rawVal !== undefined && rawVal !== "" && String(rawVal).trim() !== "-") {
              
              if (calculateZoneColor(pKey, rawVal, loc.name) === "Red") violationCount++;
              
              let cleanStr = String(rawVal).trim().toLowerCase();
              if (cleanStr === "bdl" || cleanStr === "nd") cleanStr = "0";
              
              let num = parseFloat(cleanStr.replace(/[^\d.-]/g, ''));
              if (!isNaN(num)) {
                sum += num; count++;
              } else {
                textValues.push(cleanStr); 
              }
            }
          });

          let totalSamples = count + textValues.length;
          let isStrictParam = false;
          let limits = getMasterLimits(pKey, loc.name);
          if (limits && limits.strict === true) isStrictParam = true;

          let finalParamValueColor = "Grey";

          if (count > 0) {
            let avgVal = Math.round(sum / count);
            finalParamValueColor = calculateZoneColor(pKey, avgVal, loc.name);
            
            if (isStrictParam && totalSamples > 0 && (violationCount / totalSamples) >= 0.33) {
                finalParamValueColor = "Red";
            }
          } else if (textValues.length > 0) {
            finalParamValueColor = "Green";
            for (let t of textValues) {
                let tempZone = calculateZoneColor(pKey, t, loc.name);
                if (tempZone === "Red") { finalParamValueColor = "Red"; break; }
                if (tempZone === "Yellow" && finalParamValueColor !== "Red") finalParamValueColor = "Yellow";
            }
            if (isStrictParam && totalSamples > 0 && (violationCount / totalSamples) >= 0.33) {
                finalParamValueColor = "Red";
            }
          }

          if (count > 0 || textValues.length > 0) hasDataPoints = true;

          if (finalParamValueColor === "Red") finalLocColor = "Red";
          else if (finalParamValueColor === "Yellow" && finalLocColor !== "Red") finalLocColor = "Yellow";
          else if (finalParamValueColor === "Green" && finalLocColor === "Grey") finalLocColor = "Green";
        });
      }

      if (hasDataPoints) {
        if (finalLocColor === "Red") redLocations.add(loc.name);

function doPost(e) {
  try {
    const request = JSON.parse(e.postData.contents);
    let responseData;

    // Route 1: Load Map Data
    if (request.action === "getDashboardData") {
      responseData = getDashboardData();
    } 
    // Route 2: Talk to AI
    else if (request.action === "handleChatbotQuery") {
      responseData = handleChatbotQuery(request.query, request.chatMemory, request.activeTestingType);
    }

    // Send data back to the website
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: responseData }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
